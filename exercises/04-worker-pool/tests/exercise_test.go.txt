package exercise

import (
	"reflect"
	"sync"
	"sync/atomic"
	"testing"
	"time"
)

func TestMapBoundedOrder(t *testing.T) {
	got := MapBounded([]int{4, 1, 3}, 2, func(n int) int { return n * 10 })
	want := []int{40, 10, 30}
	if !reflect.DeepEqual(got, want) {
		t.Fatalf("got %v want %v", got, want)
	}
}
func TestMapBoundedEmptyAndZeroWorkers(t *testing.T) {
	if got := MapBounded(nil, 0, func(n int) int { return n }); len(got) != 0 {
		t.Fatalf("got %v", got)
	}
	if got := MapBounded([]int{2}, 0, func(n int) int { return n + 1 }); !reflect.DeepEqual(got, []int{3}) {
		t.Fatalf("got %v", got)
	}
}
func TestMapBoundedActuallyRunsConcurrently(t *testing.T) {
	started := make(chan struct{}, 2)
	release := make(chan struct{})
	var once sync.Once
	unblock := func() { once.Do(func() { close(release) }) }
	defer unblock()
	done := make(chan []int, 1)
	go func() {
		done <- MapBounded([]int{1, 2}, 2, func(n int) int { started <- struct{}{}; <-release; return n })
	}()
	select {
	case <-started:
	case <-time.After(2 * time.Second):
		t.Fatal("worker did not start")
	}
	select {
	case <-started:
	case <-time.After(2 * time.Second):
		t.Fatal("second worker did not overlap; implementation may be sequential")
	}
	unblock()
	select {
	case got := <-done:
		if !reflect.DeepEqual(got, []int{1, 2}) {
			t.Fatalf("got %v", got)
		}
	case <-time.After(2 * time.Second):
		t.Fatal("workers did not finish")
	}
}
func TestMapBoundedNeverExceedsWorkerLimit(t *testing.T) {
	started := make(chan struct{}, 128)
	release := make(chan struct{})
	var once sync.Once
	unblock := func() { once.Do(func() { close(release) }) }
	defer unblock()
	var active, maximum atomic.Int32
	done := make(chan []int, 1)
	go func() {
		done <- MapBounded(make([]int, 128), 3, func(n int) int {
			now := active.Add(1)
			defer active.Add(-1)
			for previous := maximum.Load(); now > previous; previous = maximum.Load() {
				if maximum.CompareAndSwap(previous, now) {
					break
				}
			}
			started <- struct{}{}
			<-release
			return 1
		})
	}()
	for i := 0; i < 3; i++ {
		select {
		case <-started:
		case <-time.After(2 * time.Second):
			t.Fatal("workers did not reach their configured concurrency")
		}
	}
	// Keep all workers occupied: a goroutine per item would admit more work.
	select {
	case <-started:
		t.Fatal("more than three callbacks ran while all worker slots were occupied")
	case <-time.After(100 * time.Millisecond):
	}
	unblock()
	select {
	case got := <-done:
		if len(got) != 128 || maximum.Load() > 3 {
			t.Fatalf("results=%d maximum workers=%d", len(got), maximum.Load())
		}
	case <-time.After(2 * time.Second):
		t.Fatal("workers did not finish")
	}
}
