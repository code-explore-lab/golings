package exercise

import "sync"

type job struct{ index, value int }

func MapBounded(inputs []int, workers int, transform func(int) int) []int {
	results := make([]int, len(inputs))
	if len(inputs) == 0 {
		return results
	}
	if workers < 1 {
		workers = 1
	}
	if workers > len(inputs) {
		workers = len(inputs)
	}
	jobs := make(chan job, len(inputs))
	var wg sync.WaitGroup
	for i := 0; i < workers; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for j := range jobs {
				results[j.index] = transform(j.value)
			}
		}()
	}
	for i, v := range inputs {
		jobs <- job{i, v}
	}
	close(jobs)
	wg.Wait()
	return results
}
