package main

import (
	"log"
	"net/http"
	"time"

	exercise "gopath/exercise/14-prometheus-metrics"
)

func main() {
	metrics := exercise.NewMetrics()
	mux := http.NewServeMux()
	mux.Handle("/metrics", metrics.Handler())
	mux.Handle("/work", metrics.Instrument("/work", http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "text/plain; charset=utf-8")
		_, _ = w.Write([]byte("work\n"))
	})))
	server := &http.Server{Addr: ":2112", Handler: mux, ReadHeaderTimeout: 5 * time.Second}
	log.Fatal(server.ListenAndServe())
}
