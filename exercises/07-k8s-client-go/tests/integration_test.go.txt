package lesson

import (
	"context"
	"fmt"
	"os"
	"strings"
	"testing"
	"time"

	corev1 "k8s.io/api/core/v1"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"

	"k8s.io/client-go/tools/clientcmd"
)

// This helper refuses existing namespaces, including a GOPATH_NAMESPACE value:
// it may only remove a namespace that this test itself created successfully.
func integrationNamespace(t *testing.T) (context.Context, kubernetes.Interface, string) {
	t.Helper()
	if os.Getenv("GOPATH_INTEGRATION") != "1" {
		t.Fatal("run gopath check 07-k8s-client-go --integration, or set GOPATH_INTEGRATION=1 explicitly")
	}
	loading := clientcmd.NewDefaultClientConfigLoadingRules()
	raw, err := loading.Load()
	if err != nil {
		t.Fatalf("read kubeconfig: %v; start minikube and configure KUBECONFIG", err)
	}
	if raw.CurrentContext != "minikube" || raw.Contexts["minikube"] == nil {
		t.Fatal("integration requires kubeconfig current-context=minikube; run kubectl config use-context minikube")
	}
	config, err := clientcmd.NewNonInteractiveClientConfig(*raw, "minikube", &clientcmd.ConfigOverrides{}, loading).ClientConfig()
	if err != nil {
		t.Fatal(err)
	}
	config.Timeout = 15 * time.Second
	client, err := kubernetes.NewForConfig(config)
	if err != nil {
		t.Fatal(err)
	}
	ctx, cancel := context.WithTimeout(context.Background(), 60*time.Second)
	t.Cleanup(cancel)
	name := os.Getenv("GOPATH_NAMESPACE")
	if name != "" && !strings.HasPrefix(name, "gopath-") {
		t.Fatal("GOPATH_NAMESPACE must start with gopath- and must not exist yet")
	}
	ns := &corev1.Namespace{ObjectMeta: metav1.ObjectMeta{Name: name}}
	if name == "" {
		ns.GenerateName = "gopath-07-"
	}
	created, err := client.CoreV1().Namespaces().Create(ctx, ns, metav1.CreateOptions{})
	if err != nil {
		t.Fatalf("create isolated test namespace (never reuse an existing namespace): %v", err)
	}
	t.Cleanup(func() {
		cleanup, stop := context.WithTimeout(context.Background(), 20*time.Second)
		defer stop()
		if err := client.CoreV1().Namespaces().Delete(cleanup, created.Name, metav1.DeleteOptions{
			Preconditions: &metav1.Preconditions{UID: &created.UID},
		}); err != nil {
			t.Errorf("cleanup only test-created namespace %s: %v", created.Name, err)
		}
	})
	return ctx, client, created.Name
}

func TestIntegrationPodListWatchOnMinikube(t *testing.T) {
	ctx, client, namespace := integrationNamespace(t)
	observerCtx, stop := context.WithCancel(ctx)
	defer stop()
	events := make(chan string, 32)
	done := make(chan error, 1)
	go func() {
		done <- ObservePods(observerCtx, client, namespace, func(kind string, pod *corev1.Pod) {
			if pod.Name == "observed" {
				select {
				case events <- kind:
				case <-observerCtx.Done():
				}
			}
		})
	}()
	_, err := client.CoreV1().Pods(namespace).Create(ctx, &corev1.Pod{
		ObjectMeta: metav1.ObjectMeta{Name: "observed"},
		Spec:       corev1.PodSpec{Containers: []corev1.Container{{Name: "pause", Image: "registry.k8s.io/pause:3.10"}}},
	}, metav1.CreateOptions{})
	if err != nil {
		t.Fatal(err)
	}
	waitFor := func(want string) {
		t.Helper()
		for {
			select {
			case got := <-events:
				if got == want {
					return
				}
			case err := <-done:
				t.Fatalf("observer ended before %s: %v", want, err)
			case <-ctx.Done():
				t.Fatal(fmt.Errorf("missing real API %s event: %w", want, ctx.Err()))
			}
		}
	}
	waitFor("add")
	pods, err := ListPods(ctx, client, namespace)
	if err != nil || len(pods) != 1 || pods[0].Name != "observed" {
		t.Fatalf("real ListPods: pods=%v error=%v", pods, err)
	}
	zero := int64(0)
	if err := client.CoreV1().Pods(namespace).Delete(ctx, "observed", metav1.DeleteOptions{GracePeriodSeconds: &zero}); err != nil {
		t.Fatal(err)
	}
	waitFor("delete")
	stop()
	select {
	case err := <-done:
		if err != nil {
			t.Fatal(err)
		}
	case <-ctx.Done():
		t.Fatal("Pod observer did not stop")
	}
}
