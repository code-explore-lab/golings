// ClientFromKubeconfig is useful in a command-line program. Integration tests
// additionally verify that the selected kubeconfig context is exactly minikube.
func ClientFromKubeconfig(path string) (kubernetes.Interface, error) {
	config, err := clientcmd.BuildConfigFromFlags("", path)
	if err != nil {
		return nil, fmt.Errorf("load kubeconfig: %w", err)
	}
	return kubernetes.NewForConfig(config)
}

// podFromEvent handles a real object and the tombstone a Delete handler can
// receive when the watch missed the final state. Never mutate informer objects.
func podFromEvent(obj interface{}) (*corev1.Pod, bool) {
	switch value := obj.(type) {
	case *corev1.Pod:
		if value != nil {
			return value.DeepCopy(), true
		}
		// TODO: a Delete event may contain cache.DeletedFinalStateUnknown.
		// Unwrap both its value and pointer forms, then return a copied Pod.
	}
	return nil, false
}

func podEventHandlers(onEvent func(string, *corev1.Pod)) cache.ResourceEventHandlerFuncs {
	emit := func(kind string, obj interface{}) {
		if pod, ok := podFromEvent(obj); ok {
			onEvent(kind, pod)
		}
	}
	return cache.ResourceEventHandlerFuncs{
		AddFunc:    func(obj interface{}) { emit("add", obj) },
		UpdateFunc: func(_, current interface{}) { emit("update", current) },
		DeleteFunc: func(obj interface{}) { emit("delete", obj) },
	}
}

// ObservePods performs the initial List, then keeps a Watch and local cache.
// Callbacks must return promptly; work requiring retries belongs in a queue.
func ObservePods(ctx context.Context, client kubernetes.Interface, namespace string, onEvent func(string, *corev1.Pod)) error {
	if namespace == "" || onEvent == nil {
		return fmt.Errorf("a namespace and event callback are required")
	}
	informer := coreinformers.NewFilteredPodInformer(client, namespace, 0,
		cache.Indexers{cache.NamespaceIndex: cache.MetaNamespaceIndexFunc}, nil)
	if _, err := informer.AddEventHandler(podEventHandlers(onEvent)); err != nil {
		return fmt.Errorf("register handler: %w", err)
	}
	done := make(chan struct{})
	go func() {
		defer close(done)
		informer.Run(ctx.Done())
	}()
	if !cache.WaitForCacheSync(ctx.Done(), informer.HasSynced) {
		<-done
		// Events can arrive before WaitForCacheSync's next polling tick.
		if informer.HasSynced() {
			return nil
		}
		return fmt.Errorf("sync Pod cache: %w", ctx.Err())
	}
	<-ctx.Done()
	<-done
	return nil
}
