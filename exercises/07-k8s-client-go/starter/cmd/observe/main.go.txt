package main

import (
	"context"
	"flag"
	"log"
	"os"
	"os/signal"

	"lesson"
	"lesson/cmd/internal/cluster"

	corev1 "k8s.io/api/core/v1"
)

func main() {
	namespace := flag.String("namespace", "", "existing gopath-* namespace on minikube")
	flag.Parse()
	client, _, err := cluster.Clients(*namespace)
	if err != nil {
		log.Fatal(err)
	}
	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt)
	defer stop()
	pods, err := lesson.ListPods(ctx, client, *namespace)
	if err != nil {
		log.Fatal(err)
	}
	log.Printf("initial API List: %d Pods", len(pods))
	if err := lesson.ObservePods(ctx, client, *namespace, func(kind string, pod *corev1.Pod) {
		log.Printf("%s Pod %s/%s phase=%s", kind, pod.Namespace, pod.Name, pod.Status.Phase)
	}); err != nil {
		log.Fatal(err)
	}
}
