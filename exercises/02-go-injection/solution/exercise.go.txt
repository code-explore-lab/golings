package exercise

import (
	"errors"
	"fmt"
)

type Notifier interface{ Send(message string) error }
type Service struct{ notifier Notifier }

func NewService(n Notifier) *Service { return &Service{notifier: n} }
func (s *Service) SendWelcome(name string) error {
	if s == nil || s.notifier == nil {
		return errors.New("notifier is required")
	}
	return s.notifier.Send(fmt.Sprintf("Welcome, %s!", name))
}
