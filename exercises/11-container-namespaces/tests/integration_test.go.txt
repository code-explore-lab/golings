//go:build integration

package exercise

import (
	"os"
	"path/filepath"
	"runtime"
	"strconv"
	"testing"
)

func TestIntegrationLiveProcNamespaces(t *testing.T) {
	if runtime.GOOS != "linux" {
		t.Fatal("namespace integration requires Linux: use a Linux/WSL2 workspace; ordinary fixture tests remain cross-platform")
	}
	self, err := ReadNamespaces("/proc/self/ns")
	if err != nil {
		t.Fatalf("inspect live procfs: %v; ensure /proc is mounted", err)
	}
	pid, err := ReadNamespaces(filepath.Join("/proc", strconv.Itoa(os.Getpid()), "ns"))
	if err != nil {
		t.Fatal(err)
	}
	for _, name := range []string{"pid", "net", "uts", "mnt"} {
		a, ok := self[name]
		if !ok || a.Kind != name || a.Inode == 0 {
			t.Fatalf("missing or invalid live namespace %s: %#v", name, a)
		}
		b, ok := pid[name]
		if !ok || !SameNamespace(a, b) {
			t.Fatalf("/proc/self and /proc/%d should identify the same %s namespace", os.Getpid(), name)
		}
		t.Logf("live %s namespace inode=%d", name, a.Inode)
	}
}
