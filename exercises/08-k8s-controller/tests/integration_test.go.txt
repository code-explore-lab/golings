// This helper refuses existing namespaces, including a GOPATH_NAMESPACE value:
// it may only remove a namespace that this test itself created successfully.
func integrationNamespace(t *testing.T) (context.Context, kubernetes.Interface, string) {
	t.Helper()
	if os.Getenv("GOPATH_INTEGRATION") != "1" {
		t.Fatal("run gopath check 08-k8s-controller --integration, or set GOPATH_INTEGRATION=1 explicitly")
	}
	loading := clientcmd.NewDefaultClientConfigLoadingRules()
	raw, err := loading.Load()
	if err != nil {
		t.Fatalf("read kubeconfig: %v; start minikube and configure KUBECONFIG", err)
	}
	if raw.CurrentContext != "minikube" || raw.Contexts["minikube"] == nil {
		t.Fatal("integration requires kubeconfig current-context=minikube; run kubectl config use-context minikube")
	}
	config, err := clientcmd.NewNonInteractiveClientConfig(*raw, "minikube", &clientcmd.ConfigOverrides{}, loading).ClientConfig()
	if err != nil {
		t.Fatal(err)
	}
	config.Timeout = 15 * time.Second
	client, err := kubernetes.NewForConfig(config)
	if err != nil {
		t.Fatal(err)
	}
	ctx, cancel := context.WithTimeout(context.Background(), 60*time.Second)
	t.Cleanup(cancel)
	name := os.Getenv("GOPATH_NAMESPACE")
	if name != "" && !strings.HasPrefix(name, "gopath-") {
		t.Fatal("GOPATH_NAMESPACE must start with gopath- and must not exist yet")
	}
	ns := &corev1.Namespace{ObjectMeta: metav1.ObjectMeta{Name: name}}
	if name == "" {
		ns.GenerateName = "gopath-08-"
	}
	created, err := client.CoreV1().Namespaces().Create(ctx, ns, metav1.CreateOptions{})
	if err != nil {
		t.Fatalf("create isolated test namespace (never reuse an existing namespace): %v", err)
	}
	t.Cleanup(func() {
		cleanup, stop := context.WithTimeout(context.Background(), 20*time.Second)
		defer stop()
		if err := client.CoreV1().Namespaces().Delete(cleanup, created.Name, metav1.DeleteOptions{
			Preconditions: &metav1.Preconditions{UID: &created.UID},
		}); err != nil {
			t.Errorf("cleanup only test-created namespace %s: %v", created.Name, err)
		}
	})
	return ctx, client, created.Name
}

func TestIntegrationGadgetControllerOnMinikube(t *testing.T) {
	ctx, client, namespace := integrationNamespace(t)
	loading := clientcmd.NewDefaultClientConfigLoadingRules()
	config, err := clientcmd.NewNonInteractiveDeferredLoadingClientConfig(loading,
		&clientcmd.ConfigOverrides{CurrentContext: "minikube"}).ClientConfig()
	if err != nil {
		t.Fatal(err)
	}
	config.Timeout = 15 * time.Second
	dc, err := dynamic.NewForConfig(config)
	if err != nil {
		t.Fatal(err)
	}
	if _, err := dc.Resource(gadgetGVR).Namespace(namespace).List(ctx, metav1.ListOptions{}); err != nil {
		t.Fatalf("Gadget API unavailable: %v; first run kubectl --context minikube apply -f crd.yaml and wait for Established", err)
	}
	controllerCtx, stop := context.WithCancel(ctx)
	done := make(chan error, 1)
	go func() {
		done <- RunController(controllerCtx, dc, client, namespace, func(err error) { t.Logf("retry: %v", err) })
	}()
	t.Cleanup(func() {
		stop()
		select {
		case err := <-done:
			if err != nil {
				t.Errorf("controller shutdown: %v", err)
			}
		case <-time.After(5 * time.Second):
			t.Error("controller did not shut down")
		}
	})
	object, err := dc.Resource(gadgetGVR).Namespace(namespace).Create(ctx,
		gadget(namespace, "demo", "first"), metav1.CreateOptions{})
	if err != nil {
		t.Fatal(err)
	}
	waitForMessage := func(want string) *corev1.ConfigMap {
		t.Helper()
		tick := time.NewTicker(100 * time.Millisecond)
		defer tick.Stop()
		for {
			cm, err := client.CoreV1().ConfigMaps(namespace).Get(ctx, "demo", metav1.GetOptions{})
			if err == nil && cm.Data["message"] == want {
				return cm
			}
			if err != nil && !apierrors.IsNotFound(err) {
				t.Fatal(err)
			}
			select {
			case <-ctx.Done():
				t.Fatalf("controller did not converge message to %q: %v", want, ctx.Err())
			case <-tick.C:
			}
		}
	}
	first := waitForMessage("first")
	if err := Reconcile(ctx, dc, client, types.NamespacedName{Namespace: namespace, Name: "demo"}); err != nil {
		t.Fatal(err)
	}
	unchanged, err := client.CoreV1().ConfigMaps(namespace).Get(ctx, "demo", metav1.GetOptions{})
	if err != nil || unchanged.ResourceVersion != first.ResourceVersion {
		t.Fatalf("an idempotent reconcile must make no API write: before=%s object=%v error=%v", first.ResourceVersion, unchanged, err)
	}
	object.Object["spec"] = map[string]interface{}{"message": "second"}
	if _, err := dc.Resource(gadgetGVR).Namespace(namespace).Update(ctx, object, metav1.UpdateOptions{}); err != nil {
		t.Fatal(err)
	}
	waitForMessage("second")
}
