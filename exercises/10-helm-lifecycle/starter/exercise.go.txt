package lesson

import (
	"context"
	"fmt"
	"os/exec"
	"regexp"
	"strconv"
	"strings"
)

// Runner accepts argv, not a shell command string, so spaces stay in one argument.
type Runner func(context.Context, string, ...string) error

func ExecRunner(ctx context.Context, binary string, args ...string) error {
	output, err := exec.CommandContext(ctx, binary, args...).CombinedOutput()
	if err != nil {
		return fmt.Errorf("%s %v: %w\n%s", binary, args, err, output)
	}
	return nil
}

// This learning wrapper deliberately targets a disposable minikube environment.
// The caller creates its namespace; Helm must never choose the default context.
type Helm struct {
	Run         Runner
	Binary      string
	Namespace   string
	KubeContext string
}

var dnsName = regexp.MustCompile("^[a-z0-9]([-a-z0-9]*[a-z0-9])?$")

func validateName(kind, name string, limit int) error {
	if len(name) == 0 || len(name) > limit || !dnsName.MatchString(name) {
		return fmt.Errorf("%s must be a valid DNS label of at most %d characters", kind, limit)
	}
	return nil
}
func validatePath(kind, path string) error {
	if strings.TrimSpace(path) == "" || strings.HasPrefix(path, "-") {
		return fmt.Errorf("%s must be a nonempty path, not an option", kind)
	}
	return nil
}
func (h Helm) scope() ([]string, error) {
	if h.KubeContext != "minikube" {
		return nil, fmt.Errorf("set KubeContext to minikube explicitly")
	}
	if err := validateName("namespace", h.Namespace, 63); err != nil {
		return nil, err
	}
	return []string{"--namespace", h.Namespace, "--kube-context", h.KubeContext, "--wait", "--timeout", "30s"}, nil
}

func (h Helm) Package(ctx context.Context, chart, destination string) error {
	if err := validatePath("chart", chart); err != nil {
		return err
	}
	if err := validatePath("destination", destination); err != nil {
		return err
	}
	return h.command(ctx, "package", chart, "--destination", destination)
}
func (h Helm) Install(ctx context.Context, release, chart string) error {
	if err := validateName("release", release, 53); err != nil {
		return err
	}
	if err := validatePath("chart", chart); err != nil {
		return err
	}
	scope, err := h.scope()
	if err != nil {
		return err
	}
	return h.command(ctx, append([]string{"install", release, chart}, scope...)...)
}
func (h Helm) Upgrade(ctx context.Context, release, chart string, replicas int) error {
	if err := validateName("release", release, 53); err != nil {
		return err
	}
	if err := validatePath("chart", chart); err != nil {
		return err
	}
	if replicas < 0 {
		return fmt.Errorf("replica count must be nonnegative")
	}
	scope, err := h.scope()
	if err != nil {
		return err
	}
	args := []string{"upgrade", release, chart, "--set", "replicaCount=" + strconv.Itoa(replicas)}
	return h.command(ctx, append(args, scope...)...)
}
func (h Helm) Rollback(ctx context.Context, release string, revision int) error {
	if err := validateName("release", release, 53); err != nil {
		return err
	}
	if revision < 1 {
		return fmt.Errorf("choose an explicit revision greater than zero from helm history")
	}
	scope, err := h.scope()
	if err != nil {
		return err
	}
	return h.command(ctx, append([]string{"rollback", release, strconv.Itoa(revision)}, scope...)...)
}
