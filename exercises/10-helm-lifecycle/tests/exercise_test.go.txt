func TestHelmLifecycleBuildsExactArgv(t *testing.T) {
	scope := []string{"--namespace", "gopath-test", "--kube-context", "minikube", "--wait", "--timeout", "30s"}
	tests := []struct {
		name string
		call func(Helm, context.Context) error
		args []string
	}{
		{"package", func(h Helm, ctx context.Context) error { return h.Package(ctx, "chart with spaces", "dist folder") },
			[]string{"package", "chart with spaces", "--destination", "dist folder"}},
		{"install", func(h Helm, ctx context.Context) error { return h.Install(ctx, "demo", "dist/chart.tgz") },
			append([]string{"install", "demo", "dist/chart.tgz"}, scope...)},
		{"upgrade", func(h Helm, ctx context.Context) error { return h.Upgrade(ctx, "demo", "chart", 2) },
			append([]string{"upgrade", "demo", "chart", "--set", "replicaCount=2"}, scope...)},
		{"rollback", func(h Helm, ctx context.Context) error { return h.Rollback(ctx, "demo", 1) },
			append([]string{"rollback", "demo", "1"}, scope...)},
	}
	for _, test := range tests {
		t.Run(test.name, func(t *testing.T) {
			for _, configuredBinary := range []string{"", "tools/helm-custom"} {
				ctx, cancel := context.WithCancel(context.Background())
				defer cancel()
				var calls int
				h := Helm{Binary: configuredBinary, Namespace: "gopath-test", KubeContext: "minikube"}
				h.Run = func(gotContext context.Context, binary string, args ...string) error {
					calls++
					wantBinary := configuredBinary
					if wantBinary == "" {
						wantBinary = "helm"
					}
					if gotContext != ctx || binary != wantBinary || !reflect.DeepEqual(args, test.args) {
						t.Errorf("want context + %s %#v; got %s %#v", wantBinary, test.args, binary, args)
					}
					return nil
				}
				if err := test.call(h, ctx); err != nil {
					t.Fatal(err)
				}
				if calls != 1 {
					t.Fatalf("must call Runner exactly once, called %d times", calls)
				}
			}
		})
	}
}

func TestHelmRejectsInvalidInputBeforeRunner(t *testing.T) {
	tests := []struct {
		name string
		call func(Helm) error
	}{
		{"no chart", func(h Helm) error { return h.Package(context.Background(), "", "dist") }},
		{"chart option", func(h Helm) error { return h.Package(context.Background(), "--bad", "dist") }},
		{"no destination", func(h Helm) error { return h.Package(context.Background(), "chart", " ") }},
		{"release", func(h Helm) error { return h.Install(context.Background(), "-not-a-release", "chart") }},
		{"missing release", func(h Helm) error { return h.Install(context.Background(), "", "chart") }},
		{"missing install chart", func(h Helm) error { return h.Install(context.Background(), "demo", "") }},
		{"missing upgrade chart", func(h Helm) error { return h.Upgrade(context.Background(), "demo", "", 1) }},
		{"negative replicas", func(h Helm) error { return h.Upgrade(context.Background(), "demo", "chart", -1) }},
		{"zero revision", func(h Helm) error { return h.Rollback(context.Background(), "demo", 0) }},
		{"negative revision", func(h Helm) error { return h.Rollback(context.Background(), "demo", -1) }},
		{"unknown operation", func(h Helm) error { return h.command(context.Background(), "delete-everything") }},
		{"no operation", func(h Helm) error { return h.command(context.Background()) }},
		{"empty argument", func(h Helm) error { return h.command(context.Background(), "package", "") }},
		{"implicit context", func(h Helm) error { h.KubeContext = ""; return h.Install(context.Background(), "demo", "chart") }},
		{"another cluster", func(h Helm) error {
			h.KubeContext = "production"
			return h.Upgrade(context.Background(), "demo", "chart", 1)
		}},
		{"no namespace", func(h Helm) error { h.Namespace = ""; return h.Rollback(context.Background(), "demo", 1) }},
		{"invalid namespace", func(h Helm) error {
			h.Namespace = "Name Space"
			return h.Install(context.Background(), "demo", "chart")
		}},
	}
	for _, test := range tests {
		t.Run(test.name, func(t *testing.T) {
			calls := 0
			h := Helm{Namespace: "gopath-test", KubeContext: "minikube", Run: func(context.Context, string, ...string) error {
				calls++
				return nil
			}}
			if err := test.call(h); err == nil {
				t.Fatal("invalid input must return an error")
			}
			if calls != 0 {
				t.Fatal("invalid input must not execute Helm")
			}
		})
	}
}

func TestHelmPreservesErrorsAndCancellation(t *testing.T) {
	want := errors.New("helm failed")
	h := Helm{Namespace: "gopath-test", KubeContext: "minikube", Run: func(context.Context, string, ...string) error { return want }}
	calls := []func() error{
		func() error { return h.Package(context.Background(), "chart", "dist") },
		func() error { return h.Install(context.Background(), "demo", "chart") },
		func() error { return h.Upgrade(context.Background(), "demo", "chart", 0) },
		func() error { return h.Rollback(context.Background(), "demo", 1) },
	}
	for _, call := range calls {
		if err := call(); !errors.Is(err, want) {
			t.Fatalf("Runner error must remain inspectable: %v", err)
		}
	}
	h.Run = nil
	if err := h.Package(context.Background(), "chart", "dist"); err == nil {
		t.Fatal("nil Runner must produce a configuration error")
	}
	ctx, cancel := context.WithCancel(context.Background())
	cancel()
	h.Run = func(context.Context, string, ...string) error {
		t.Fatal("canceled context executed Runner")
		return nil
	}
	if err := h.Package(ctx, "chart", "dist"); !errors.Is(err, context.Canceled) {
		t.Fatalf("cancellation must be returned before execution: %v", err)
	}
}

// A child copy of this test binary provides a portable external process.
// It never invokes an OS shell, Helm, kubectl, or a cluster.
func TestRunnerHelperProcess(t *testing.T) {
	if os.Getenv("GOPATH_TEST_HELPER") != "1" {
		return
	}
	for _, arg := range os.Args {
		if arg == "fail-helper" {
			fmt.Fprint(os.Stderr, "synthetic failure detail")
			os.Exit(7)
		}
	}
	os.Exit(0)
}

func TestExecRunnerCapturesProcessFailure(t *testing.T) {
	t.Setenv("GOPATH_TEST_HELPER", "1")
	binary, err := os.Executable()
	if err != nil {
		t.Fatal(err)
	}
	if err := ExecRunner(context.Background(), binary, "-test.run=^TestRunnerHelperProcess$", "--", "ok-helper"); err != nil {
		t.Fatalf("successful process: %v", err)
	}
	err = ExecRunner(context.Background(), binary, "-test.run=^TestRunnerHelperProcess$", "--", "fail-helper")
	var exit *exec.ExitError
	if !errors.As(err, &exit) || exit.ExitCode() != 7 || !strings.Contains(err.Error(), "synthetic failure detail") {
		t.Fatalf("must preserve process failure and stderr: %v", err)
	}
}
